
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA

# Step 1: Load the Wine Dataset
url = "https://media.geeksforgeeks.org/wp-content/uploads/Wine.csv"
df = pd.read_csv(url)

print("Original Dataset:")
print(df.head())
print("\nDataset Shape:", df.shape)

# Step 2: Separate features and target variable
# The Wine.csv dataset uses 'Customer_Segment' as its class label.
X = df.drop("Customer_Segment", axis=1)
y = df["Customer_Segment"]

# Step 3: Standardize the features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Step 4: Apply PCA
pca = PCA(n_components=2)
X_pca = pca.fit_transform(X_scaled)

# Step 5: Create a new DataFrame
pca_df = pd.DataFrame(
    X_pca, columns=["PC1", "PC2"]
)
pca_df["Wine_Class"] = y.values

print("\nDataset After PCA:")
print(pca_df.head())

# Step 6: Explained variance
print("\nExplained Variance Ratio:")
print(pca.explained_variance_ratio_)

print("\nTotal Variance Explained:",
      sum(pca.explained_variance_ratio_) * 100, "%")

# Step 7: Visualize the PCA results
plt.figure(figsize=(8, 6))

for wine_class in sorted(pca_df["Wine_Class"].unique()):
    subset = pca_df[pca_df["Wine_Class"] == wine_class]

    plt.scatter(
        subset["PC1"],
        subset["PC2"],
        label=f"Wine Class {wine_class}",
        alpha=0.7
    )

plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.title("PCA on Wine Dataset")
plt.legend()
plt.grid(True)
plt.show()
