
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Step 1: Load Dataset
df = pd.read_csv("City_Air_Quality.csv")

print("Original Dataset:")
print(df.head())
print("\nDataset Shape:", df.shape)
print("\nColumns:", df.columns.tolist())
print("\nMissing Values:\n", df.isnull().sum())

# Step 2: Standardize Column Names
df.columns = (
    df.columns.str.strip()
    .str.lower()
    .str.replace(r"[^a-z0-9]+", "", regex=True)
)

# Step 3: Identify Date and AQI Columns
date_col = "date"
aqi_col = "aqi"

required = [date_col, aqi_col]
missing = [col for col in required if col not in df.columns]

if missing:
    raise ValueError(f"Missing required columns: {missing}")

# Step 4: Data Cleaning
df["date"] = pd.to_datetime(df["date"], errors="coerce")

pollutants = [
    col for col in ["pm25", "pm10", "co"]
    if col in df.columns
]

for col in ["aqi"] + pollutants:
    df[col] = pd.to_numeric(df[col], errors="coerce")
    df.loc[df[col] < 0, col] = np.nan

df = df.dropna(subset=["date", "aqi"])
df = df.sort_values("date")

# Fill missing pollutant values
for col in pollutants:
    df[col] = df[col].fillna(df[col].median())

# Step 5: Daily Aggregation
daily = df.groupby(df["date"].dt.date).agg(
    {"aqi": "mean", **{p: "mean" for p in pollutants}}
)

daily.index = pd.to_datetime(daily.index)

print("\nAQI Statistics:")
print(df["aqi"].describe())

print("\nAverage Daily AQI:")
print(daily.head())

# Step 6: AQI Time Series Plot
plt.figure(figsize=(10, 5))
plt.plot(daily.index, daily["aqi"], color="blue")
plt.title("AQI Trend Over Time")
plt.xlabel("Date")
plt.ylabel("Average AQI")
plt.grid(True)
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

# Step 7: Individual Pollutant Trends
for pollutant in pollutants:
    plt.figure(figsize=(10, 4))
    plt.plot(
        daily.index,
        daily[pollutant],
        label=pollutant.upper(),
        color="green"
    )
    plt.title(f"{pollutant.upper()} Trend Over Time")
    plt.xlabel("Date")
    plt.ylabel("Pollutant Concentration")
    plt.legend()
    plt.grid(True)
    plt.xticks(rotation=45)
    plt.tight_layout()
    plt.show()

# Step 8: Monthly AQI Bar Plot
monthly = df.groupby(
    df["date"].dt.to_period("M")
)["aqi"].mean()

plt.figure(figsize=(9, 5))
monthly.plot(kind="bar", color="orange")
plt.title("Average Monthly AQI")
plt.xlabel("Month")
plt.ylabel("Average AQI")
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

# Step 9: AQI Category Distribution
# Categories use the Indian AQI index ranges
def get_category(aqi):
    if aqi <= 50:
        return "Good"
    elif aqi <= 100:
        return "Satisfactory"
    elif aqi <= 200:
        return "Moderate"
    elif aqi <= 300:
        return "Poor"
    elif aqi <= 400:
        return "Very Poor"
    return "Severe"

df["AQI_Category"] = df["aqi"].apply(get_category)

categories = [
    "Good", "Satisfactory", "Moderate",
    "Poor", "Very Poor", "Severe"
]

category_data = [
    df.loc[df["AQI_Category"] == c, "aqi"]
    for c in categories
]

# Step 10: Box Plot
plt.figure(figsize=(10, 5))
plt.boxplot(
    category_data,
    tick_labels=categories
)
plt.title("AQI Distribution by Category")
plt.xlabel("AQI Category")
plt.ylabel("AQI")
plt.xticks(rotation=30)
plt.tight_layout()
plt.show()

# Step 11: Scatter Plots - AQI vs Pollutants
for pollutant in pollutants:
    plt.figure(figsize=(7, 5))
    plt.scatter(
        df[pollutant],
        df["aqi"],
        alpha=0.5,
        color="purple"
    )
    plt.title(f"AQI vs {pollutant.upper()}")
    plt.xlabel(pollutant.upper())
    plt.ylabel("AQI")
    plt.grid(True)
    plt.tight_layout()
    plt.show()

# Step 12: Correlation Heatmap Using Matplotlib
numeric_cols = ["aqi"] + pollutants
corr = df[numeric_cols].corr()

fig, ax = plt.subplots(figsize=(7, 5))
im = ax.imshow(corr, cmap="coolwarm", vmin=-1, vmax=1)

ax.set_xticks(range(len(numeric_cols)))
ax.set_yticks(range(len(numeric_cols)))
ax.set_xticklabels(numeric_cols)
ax.set_yticklabels(numeric_cols)

for i in range(len(numeric_cols)):
    for j in range(len(numeric_cols)):
        ax.text(
            j, i, f"{corr.iloc[i, j]:.2f}",
            ha="center", va="center"
        )

plt.colorbar(im, label="Correlation")
plt.title("AQI and Pollutant Correlation")
plt.tight_layout()
plt.show()

# Step 13: Save Processed Dataset
df.to_csv("Cleaned_City_Air_Quality.csv", index=False)

print("\nAQI Visualization Completed Successfully!")
