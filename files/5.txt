import numpy as np

#input patterns
x1 = np.array([1,1,1,1,1,1]).reshape(6,1)
x2 = np.array([-1,-1,-1,-1,-1,-1]).reshape(6,1)
x3 = np.array([1,1,-1,-1,1,1]).reshape(6,1)
x4 = np.array([-1,1,1,1,1,-1]).reshape(6,1)


#Target patterns
y1 = np.array([1,1,1]).reshape(3,1)
y2 = np.array([-1,-1,-1]).reshape(3,1)
y3 = np.array([1,1,-1]).reshape(3,1)
y4 = np.array([-1,1,1]).reshape(3,1)


inputset = np.concatenate((x1,x2,x3,x4),axis=1)
print(inputset)

outputset = np.concatenate((y1,y2,y3,y4),axis=1)
print(outputset)


weight = np.dot(inputset,outputset.T)

print(weight)

#Testing input patterns

def test_input(x,weight):

    y=np.dot(weight.T,x)

    
    y[y<0] = -1
    y[y>0] = 1

    return np.array(y)

print(test_input(x1,weight))

def test_targets(y,weight):

    x = np.dot(weight,y)


    x[x<0] = -1
    x[x>0] = 1

    return np.array(x)

print(test_targets(y1,weight))
